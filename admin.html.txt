    <div id="dashboardSection" style="display: none;">
        <div class="admin-header">
            <h1>Fortinet Admin Dashboard</h1>
            <button class="btn btn-delete" onclick="logout()">Logout</button>
        </div>
        <div class="stats" id="stats"></div>
        <div class="filter-bar">
            <button class="filter-btn active" onclick="filterComplaints('all')">All</button>
            <button class="filter-btn" onclick="filterComplaints('Pending')">Pending</button>
            <button class="filter-btn" onclick="filterComplaints('Replied')">Replied</button>
            <button class="filter-btn" onclick="filterComplaints('Resolved')">Resolved</button>
            <button class="filter-btn" onclick="loadComplaints()">🔄 Refresh</button>
        </div>
        <div id="complaintsContainer">Loading...</div>
    </div>

    <script>
        const API_URL = "/api";
        let allComplaints = [];
        let currentFilter = 'all';

        function checkLogin() {
            if (localStorage.getItem('adminLoggedIn') === 'true') {
                document.getElementById('loginSection').style.display = 'none';
                document.getElementById('dashboardSection').style.display = 'block';
                loadComplaints();
            } else {
                document.getElementById('loginSection').style.display = 'block';
                document.getElementById('dashboardSection').style.display = 'none';
            }
        }

        document.getElementById('loginForm')?.addEventListener('submit', function(e) {
            e.preventDefault();
            const username = document.getElementById('username').value;
            const password = document.getElementById('password').value;
            
            if (username === 'fortinet' && password === 'fortinet123') {
                localStorage.setItem('adminLoggedIn', 'true');
                checkLogin();
            } else {
                document.getElementById('loginError').innerHTML = '<div class="alert alert-error">Invalid credentials</div>';
            }
        });

        async function loadComplaints() {
            const container = document.getElementById('complaintsContainer');
            container.innerHTML = 'Loading...';
            try {
                const res = await fetch(`${API_URL}/complaints`);
                const data = await res.json();
                if (data.complaints) {
                    allComplaints = data.complaints;
                    updateStats();
                    displayComplaints();
                }
            } catch (err) {
                container.innerHTML = '<div class="alert alert-error">Error loading</div>';
            }
        }

        function updateStats() {
            document.getElementById('stats').innerHTML = `
                <div class="stat-card"><div class="stat-number">${allComplaints.length}</div><div>Total</div></div>
                <div class="stat-card"><div class="stat-number">${allComplaints.filter(c => c.status === 'Pending').length}</div><div>Pending</div></div>
                <div class="stat-card"><div class="stat-number">${allComplaints.filter(c => c.status === 'Replied').length}</div><div>Replied</div></div>
                <div class="stat-card"><div class="stat-number">${allComplaints.filter(c => c.status === 'Resolved').length}</div><div>Resolved</div></div>
            `;
        }

        function displayComplaints() {
            let filtered = allComplaints;
            if (currentFilter !== 'all') filtered = allComplaints.filter(c => c.status === currentFilter);
            
            const container = document.getElementById('complaintsContainer');
            if (filtered.length === 0) {
                container.innerHTML = '<div class="alert alert-info">No complaints</div>';
                return;
            }
            
            container.innerHTML = '';
            filtered.forEach(c => {
                container.innerHTML += `
                    <div class="complaint-card ${c.status === 'Resolved' ? 'resolved' : ''}">
                        <p><strong>${c.name}</strong> (${c.email})</p>
                        <p><strong>Subject:</strong> ${c.subject}</p>
                        <p><strong>Message:</strong> ${c.message}</p>
                        <p><strong>Status:</strong> ${c.status}</p>
                        ${c.reply ? `<p><strong>Your Reply:</strong> ${c.reply}</p>` : ''}
                        <textarea id="reply-${c.id}" class="reply-textarea" placeholder="Type reply..."></textarea>
                        <button class="btn btn-reply" onclick="sendReply('${c.id}')">Send Reply</button>
                        ${c.status !== 'Resolved' ? `<button class="btn btn-resolve" onclick="markResolved('${c.id}')">Mark Resolved</button>` : ''}
                        <button class="btn btn-delete" onclick="deleteComplaint('${c.id}')">Delete</button>
                    </div>
                `;
            });
        }

        async function sendReply(id) {
            const reply = document.getElementById(`reply-${id}`).value;
            if (!reply) return alert('Enter reply');
            await fetch(`${API_URL}/complaints/${id}/reply`, {
                method: 'PUT',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ reply })
            });
            loadComplaints();
        }

        async function markResolved(id) {
            await fetch(`${API_URL}/complaints/${id}/resolve`, { method: 'PUT' });
            loadComplaints();
        }

        async function deleteComplaint(id) {
            if (confirm('Delete?')) {
                await fetch(`${API_URL}/complaints/${id}`, { method: 'DELETE' });
                loadComplaints();
            }
        }

        function filterComplaints(filter) {
            currentFilter = filter;
            document.querySelectorAll('.filter-btn').forEach(btn => btn.classList.remove('active'));
            event.target.classList.add('active');
            displayComplaints();
        }

        function logout() {
            localStorage.removeItem('adminLoggedIn');
            checkLogin();
        }

        checkLogin();
    </script>
